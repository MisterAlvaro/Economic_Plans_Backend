-- 1. Divisiones (ej: EMPET — DIV.MATANZAS, DIV.HABANA, etc.)
CREATE TABLE divisions (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL,  -- Nombre oficial de la división
  code VARCHAR(10) UNIQUE,            -- Código interno (ej: "MTZAS")
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Usuarios (economistas y administradores)
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,  -- Encriptado con bcrypt
  full_name VARCHAR(100) NOT NULL,      -- Ej: "Yaima Gonzalez Alvarez"
  division_id INTEGER REFERENCES divisions(id) ON DELETE CASCADE,
  role VARCHAR(20) NOT NULL CHECK (role IN ('admin', 'economist')), 
  is_active BOOLEAN DEFAULT TRUE,
  last_login TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Plan Maestro anual (definido por admin)
CREATE TABLE master_plans (
  id SERIAL PRIMARY KEY,
  year INTEGER UNIQUE NOT NULL CHECK (year BETWEEN 2000 AND 2100),
  status VARCHAR(20) NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'active')),
  file_name VARCHAR(255),
  created_by INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE
);

-- 4. Planes Económicos por división (regidos por el plan maestro anual)
CREATE TABLE economic_plans (
  id SERIAL PRIMARY KEY,
  division_id INTEGER NOT NULL REFERENCES divisions(id) ON DELETE CASCADE,
  master_plan_id INTEGER NOT NULL REFERENCES master_plans(id) ON DELETE RESTRICT,
  year INTEGER NOT NULL CHECK (year BETWEEN 2000 AND 2100),  -- Ej: 2025
  version INTEGER DEFAULT 1,                                 -- Versión del plan (si hay ediciones)
  status VARCHAR(20) CHECK (status IN ('draft', 'reviewed', 'approved')),
  created_by INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  reviewed_by INTEGER REFERENCES users(id) ON DELETE SET NULL,  -- Ej: "Félix Luis Peralta Curbelo"
  approved_by INTEGER REFERENCES users(id) ON DELETE SET NULL,  -- Ej: "Jorge Emilio Vizcon Villareal"
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE,
  UNIQUE(division_id, year)  -- Solo un plan por año/división
);

-- 5. Hojas del Plan Maestro (estructura flexible)
CREATE TABLE master_plan_sheets (
  id SERIAL PRIMARY KEY,
  master_plan_id INTEGER NOT NULL REFERENCES master_plans(id) ON DELETE CASCADE,
  sheet_name VARCHAR(50) NOT NULL,
  data JSONB NOT NULL,
  UNIQUE(master_plan_id, sheet_name)
);

-- 6. Datos de las Hojas Excel divisionales (estructura flexible)
CREATE TABLE plan_sheets (
  id SERIAL PRIMARY KEY,
  plan_id INTEGER NOT NULL REFERENCES economic_plans(id) ON DELETE CASCADE,
  sheet_name VARCHAR(50) NOT NULL,  -- Ej: "Resumen", "IndEco"
  data JSONB NOT NULL,              -- Almacena toda la hoja como JSON (filas/columnas)
  UNIQUE(plan_id, sheet_name)
);

-- 7. Celdas con Fórmulas (para recalcular dinámicamente)
CREATE TABLE formula_cells (
  id SERIAL PRIMARY KEY,
  sheet_id INTEGER NOT NULL REFERENCES plan_sheets(id) ON DELETE CASCADE,
  cell_reference VARCHAR(10) NOT NULL,  -- Ej: "D6", "F17"
  formula TEXT NOT NULL,                -- Ej: "=SUM(D8:D9)"
  depends_on TEXT[],                    -- Celdas referenciadas (ej: {"D8", "D9"})
  last_value NUMERIC(15, 2),            -- Resultado caché
  last_calculated_at TIMESTAMP WITH TIME ZONE
);

-- 8. Historial de recomendaciones IA para reajustes
CREATE TABLE ai_reajuste_history (
  id SERIAL PRIMARY KEY,
  plan_id INTEGER NOT NULL REFERENCES economic_plans(id) ON DELETE CASCADE,
  requested_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
  model_name VARCHAR(80) NOT NULL,
  context JSONB,                        -- Contexto enviado al modelo (faltantes/extras/coincidencias)
  recommendation TEXT NOT NULL,        -- Respuesta de la IA local
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. Historial de Cambios (auditoría)
CREATE TABLE plan_audit_log (
  id SERIAL PRIMARY KEY,
  plan_id INTEGER NOT NULL REFERENCES economic_plans(id) ON DELETE CASCADE,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  action VARCHAR(20) NOT NULL CHECK (action IN ('create', 'update', 'review', 'approve')),
  changed_table VARCHAR(30),            -- Ej: "plan_sheets", "formula_cells"
  changed_field VARCHAR(50),           -- Ej: "data", "formula"
  old_value TEXT,
  new_value TEXT,
  ip_address VARCHAR(45),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 10. Configuración de Indicadores (para validación)
CREATE TABLE economic_indicators (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) NOT NULL,           -- Ej: "Valor Agregado Bruto"
  code VARCHAR(30) UNIQUE NOT NULL,     -- Ej: "VAB"
  unit VARCHAR(10),                     -- Ej: "MP", "horas"
  description TEXT,
  formula_template TEXT                 -- Ej: "=+C3+C4+C5-C6+..."
);
